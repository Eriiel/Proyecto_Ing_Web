-- Crear la base de datos
CREATE DATABASE IF NOT EXISTS click_and_paw CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE click_and_paw;

-- ==========================================================================
-- TABLA: administradores
-- Para el login del panel de administración (RF5)
-- ==========================================================================
CREATE TABLE administradores (
    id INT AUTO_INCREMENT PRIMARY KEY,
    usuario VARCHAR(50) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL, -- Almacenar con password_hash()
    nombre_completo VARCHAR(100) NOT NULL,
    email VARCHAR(100) NOT NULL UNIQUE,
    fecha_creacion TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    ultimo_acceso TIMESTAMP NULL,
    activo TINYINT(1) DEFAULT 1
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ==========================================================================
-- TABLA: mascotas
-- Catálogo de mascotas disponibles para adopción (RF6, RF7, RF8, RF9)
-- ==========================================================================
CREATE TABLE mascotas (
    id INT AUTO_INCREMENT PRIMARY KEY,
    
    -- Información básica
    nombre VARCHAR(100) NOT NULL,
    especie VARCHAR(50) NOT NULL, -- Perro, Gato, Conejo, Ave, Otro
    raza VARCHAR(100),
    
    -- Características físicas
    edad INT NOT NULL, -- Edad en años
    edad_categoria ENUM('Cachorro', 'Adulto', 'Senior') NOT NULL,
    genero ENUM('Macho', 'Hembra') NOT NULL,
    tamano ENUM('Pequeño', 'Mediano', 'Grande') NOT NULL,
    color VARCHAR(50),
    
    -- Información descriptiva
    descripcion TEXT NOT NULL, -- Descripción general corta
    personalidad TEXT, -- Descripción de personalidad y temperamento
    historia TEXT, -- Historia de rescate
    
    -- Salud
    estado_salud TEXT, -- Vacunas, esterilización, tratamientos
    vacunas_al_dia TINYINT(1) DEFAULT 0,
    esterilizado TINYINT(1) DEFAULT 0,
    necesidades_especiales TEXT, -- Dietas, medicamentos, cuidados especiales
    
    -- Multimedia
    foto_principal VARCHAR(255), -- Nombre del archivo de la foto principal
    fotos_adicionales TEXT, -- JSON o lista separada por comas de fotos adicionales
    video_url VARCHAR(255), -- URL de video (YouTube, Vimeo, etc.)
    
    -- Estado y control
    estado ENUM('Disponible', 'En proceso de adopción', 'Adoptado') DEFAULT 'Disponible',
    fecha_registro TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    fecha_actualizacion TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    -- Índices para búsquedas rápidas
    INDEX idx_estado (estado),
    INDEX idx_especie (especie),
    INDEX idx_nombre (nombre)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ==========================================================================
-- TABLA: solicitudes
-- Solicitudes de adopción de los usuarios (RF3, RF10, RF11, RF12)
-- ==========================================================================
CREATE TABLE solicitudes (
    id INT AUTO_INCREMENT PRIMARY KEY,
    
    -- Relación con mascota
    mascota_id INT NOT NULL,
    
    -- Información del solicitante
    nombre_completo VARCHAR(255) NOT NULL,
    email VARCHAR(255) NOT NULL,
    telefono VARCHAR(50) NOT NULL,
    direccion TEXT NOT NULL,
    
    -- Información del hogar y motivación
    descripcion_hogar TEXT NOT NULL, -- Familia, otras mascotas, tipo de vivienda
    motivo_adopcion TEXT NOT NULL, -- Mínimo 500 caracteres
    experiencia TEXT, -- Experiencia previa con mascotas (opcional)
    
    -- Estado y seguimiento
    estado ENUM('Recibida', 'En revisión', 'Aprobada', 'Rechazada') DEFAULT 'Recibida',
    notas_admin TEXT, -- Notas internas del administrador
    
    -- Fechas
    fecha_solicitud TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    fecha_actualizacion TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    -- Clave foránea
    FOREIGN KEY (mascota_id) REFERENCES mascotas(id) ON DELETE CASCADE,
    
    -- Índices
    INDEX idx_estado (estado),
    INDEX idx_mascota (mascota_id),
    INDEX idx_fecha (fecha_solicitud)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- Vista de mascotas disponibles con información resumida
CREATE VIEW v_mascotas_disponibles AS
SELECT 
    id,
    nombre,
    especie,
    raza,
    edad,
    edad_categoria,
    genero,
    tamano,
    descripcion,
    foto_principal,
    estado
FROM mascotas
WHERE estado = 'Disponible'
ORDER BY fecha_registro DESC;

-- Vista de solicitudes pendientes
CREATE VIEW v_solicitudes_pendientes AS
SELECT 
    s.id AS solicitud_id,
    s.nombre_completo AS solicitante,
    s.email,
    s.telefono,
    m.nombre AS mascota,
    m.especie,
    s.estado,
    s.fecha_solicitud
FROM solicitudes s
INNER JOIN mascotas m ON s.mascota_id = m.id
WHERE s.estado IN ('Recibida', 'En revisión')
ORDER BY s.fecha_solicitud DESC;
